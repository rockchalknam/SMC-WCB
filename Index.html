/** * Level Arcade data feed. * Reads Google Form level-code uploads (.txt or Google Doc) and serves them * as JSON to the Level Arcade page hosted on GitHub Pages. * Paste this into Extensions > Apps Script on the form's response sheet. */ const CONFIG = { SHEET_NAME: 'Form Responses 1', // tab the form writes to DATA_FILE_NAME: 'level-arcade-data.json', // cache file the page reads PAGE_TITLE: 'Level Arcade' }; /* ---------- Sheet menu ---------- */ function onOpen() { SpreadsheetApp.getUi() .createMenu('Level Arcade') .addItem('Set up auto-update (run once)', 'setupTrigger') .addItem('Rebuild all levels now', 'rebuildAll') .addToUi(); } function setupTrigger() { const ss = SpreadsheetApp.getActive(); ScriptApp.getProjectTriggers() .filter(t => t.getHandlerFunction() === 'onSubmit') .forEach(t => ScriptApp.deleteTrigger(t)); ScriptApp.newTrigger('onSubmit').forSpreadsheet(ss).onFormSubmit().create(); const n = rebuildLevels_(false); SpreadsheetApp.getUi().alert(`Auto-update is on. ${n} levels loaded.`); } /* ---------- Triggers ---------- */ // Runs on every form submission. Only reads files it hasn't seen before. function onSubmit() { rebuildLevels_(false); } // Menu option: re-reads every file from scratch (use if a code looks wrong). function rebuildAll() { const n = rebuildLevels_(true); SpreadsheetApp.getUi().alert(`Rebuilt ${n} levels.`); } /* ---------- Core ---------- */ function rebuildLevels_(fresh) { const lock = LockService.getScriptLock(); lock.waitLock(120000); // class submits at once, so take turns try { const sheet = SpreadsheetApp.getActive().getSheetByName(CONFIG.SHEET_NAME); const rows = sheet.getDataRange().getValues(); const head = rows.shift().map(h => String(h)); const col = re => head.findIndex(h => re.test(h)); const C = { time: col(/timestamp/i), email: col(/email/i), first: col(/first/i), last: col(/last/i), period: col(/period/i), title: col(/title/i), file: col(/level code|file/i) }; if (C.first < 0 || C.file < 0) throw new Error('Could not find the First Name or file upload column.'); const old = fresh ? {} : readData_().cache || {}; const cache = {}; // fileId -> code, so we never re-read a file const byStudent = {}; // newest submission wins rows .filter(r => r[C.first] && r[C.file]) .sort((a, b) => new Date(a[C.time]) - new Date(b[C.time])) .forEach(r => { const ids = String(r[C.file]).match(/[-\w]{25,}/g); if (!ids) return; const id = ids[ids.length - 1]; // if they uploaded more than one, use the last let code = old[id]; if (code === undefined) { try { code = readCodeFile_(id); } catch (err) { code = ''; } } cache[id] = code; if (!code) return; const first = properCase_(r[C.first]); const last = C.last >= 0 ? properCase_(r[C.last]) : ''; const period = normPeriod_(C.period >= 0 ? r[C.period] : ''); const key = (C.email >= 0 && r[C.email] ? String(r[C.email]) : first + last + period).toLowerCase(); byStudent[key] = { name: first + (last ? ' ' + last[0].toUpperCase() + '.' : ''), // first name + last initial only period: period, title: C.title >= 0 ? String(r[C.title]).trim() : '', code: code }; }); const levels = Object.values(byStudent); writeData_({ updated: new Date().toISOString(), levels: levels, cache: cache }); return levels.length; } finally { lock.releaseLock(); } } // Reads a level code from a .txt upload, or from a Google Doc if a // student pasted their code into a Doc instead. function readCodeFile_(id) { const file = DriveApp.getFileById(id); let text; if (file.getMimeType() === MimeType.GOOGLE_DOCS) { text = DocumentApp.openById(id).getBody().getText() .replace(/[\u201C\u201D]/g, '"').replace(/[\u2018\u2019]/g, "'"); // undo smart quotes } else { text = file.getBlob().getDataAsString('UTF-8'); } return text.replace(/^\uFEFF/, '').trim(); } // "nAM", "NAM", " nam " -> "Nam". Handles "mary-jane" -> "Mary-Jane" and "ana maria" -> "Ana Maria". function properCase_(s) { return String(s || '').trim().replace(/\s+/g, ' ').toLowerCase() .replace(/(^|[\s\-'])(\p{L})/gu, (m, sep, ch) => sep + ch.toUpperCase()); } function normPeriod_(p) { const m = String(p).match(/\d/); const names = ['1st', '2nd', '3rd', '4th', '5th', '6th', '7th']; return m ? names[+m[0] - 1] || m[0] : (String(p).trim() || 'Other'); } /* ---------- Data file (one small JSON file in your Drive) ---------- */ function dataFile_() { const props = PropertiesService.getScriptProperties(); const id = props.getProperty('DATA_FILE_ID'); if (id) { try { return DriveApp.getFileById(id); } catch (e) {} } const f = DriveApp.createFile(CONFIG.DATA_FILE_NAME, '{"levels":[]}', 'application/json'); props.setProperty('DATA_FILE_ID', f.getId()); return f; } function readData_() { try { return JSON.parse(dataFile_().getBlob().getDataAsString()); } catch (e) { return { levels: [] }; } } function writeData_(obj) { dataFile_().setContent(JSON.stringify(obj)); } /* ---------- JSON feed for the GitHub page ---------- */ // Deploy as a Web app with access set to "Anyone". // Only names (first + last initial), periods, titles and codes are shared. No emails. function doGet() { const data = readData_(); const out = { updated: data.updated || null, levels: data.levels || [] }; return ContentService.createTextOutput(JSON.stringify(out)) .setMimeType(ContentService.MimeType.JSON); }
